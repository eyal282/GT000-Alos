let alternatingTier = 1;

function getRndInteger(min, max) {
  return Math.floor(Math.random() * (max - min) ) + min;
}

setTimeout(function() { 
    let primary_obj = []
    
    for(let abc=0;abc < 10000;abc++)
	{
		let Eyal_temp = {};

		let Eyal_key;
		
		let obj_legalOptions = [];
		
		for(let def = 0;def < obj.length;def++)
		{
			if(obj_tier[def] == alternatingTier)
			{
				obj_legalOptions.push(obj[def])
			}
		}
		
		let randomResult = getRndInteger(0, obj_legalOptions.length);

		Eyal_key = `half_weekly_shop_${obj_legalOptions[randomResult]}`
		
		Eyal_temp.StartDateTime = 1672963199 + (abc * 86400 * 3);
		Eyal_temp.EndDateTime = 1672963199 + (86400 * 3) + (abc * 86400 * 3);
		let Eyal_date = new Date(Eyal_temp.StartDateTime * 1000);

		Eyal_temp.Name = Eyal_date.toDateString();
		Eyal_temp.Name = Eyal_temp.Name.replace("Sun ", "");
		Eyal_temp.Name = Eyal_temp.Name.replace("Mon ", "");
		Eyal_temp.Name = Eyal_temp.Name.replace("Tue ", "");
		Eyal_temp.Name = Eyal_temp.Name.replace("Wed ", "");
		Eyal_temp.Name = Eyal_temp.Name.replace("Thu ", "");
		Eyal_temp.Name = Eyal_temp.Name.replace("Fri ", "");
		Eyal_temp.Name = Eyal_temp.Name.replace("Sat ", "");
		Eyal_temp.ScheduledProducts = [];

		let secondary_obj = {};

		secondary_obj.Category = "Promoted Offers"
		secondary_obj.Priority = ""
		secondary_obj.ABTest = ""
		secondary_obj.PromotionGroup = "NewPlant"
		secondary_obj.AdditionalSegment = ""
		secondary_obj.SmartKey = ""
		secondary_obj.SoftCurrency = ""
		secondary_obj.SoftCost = -1
		secondary_obj.MaxPurchase = 3
		secondary_obj.SoftCostPerPurchase = -1
		secondary_obj.Visuals = "RTID(0)"
		
		secondary_obj.Sku = [];
		secondary_obj.Sku.push(Eyal_key)
		secondary_obj.Sku.push(Eyal_key)
	   
		if(Eyal_temp.EndDateTime > 2147483647)
			break;

		Eyal_temp.ScheduledProducts.push(secondary_obj)
		
		primary_obj.push(Eyal_temp)
		
		alternatingTier++;
		
		if(alternatingTier > 5)
		{
			alternatingTier = 1;
		}
	}
    
    navigator.clipboard.writeText(JSON.stringify(primary_obj, null, "\t"))
}, 1000);
